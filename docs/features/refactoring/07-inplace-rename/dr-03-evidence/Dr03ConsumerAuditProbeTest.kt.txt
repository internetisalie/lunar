/**
 * DR-03 measurement probe. Prints observations; asserts nothing. Removed after the run.
 */
class Dr03ConsumerAuditProbeTest : BasePlatformTestCase() {
    private val fixtureText =
        """
        local counter = 0
        local function helper(alpha, beta)
          return alpha + beta + counter
        end
        local unused = 0
        counter = helper(1, 2)
        print(counter)
        """.trimIndent() + "\n"

    private fun emit(
        consumer: String,
        line: String,
    ) {
        println("DR03|$consumer|$line")
    }

    private fun probe(
        consumer: String,
        body: () -> Unit,
    ) {
        try {
            body()
        } catch (failure: Throwable) {
            emit(consumer, "THREW=${failure.javaClass.name}: ${failure.message}")
        }
    }

    private fun configure(text: String = fixtureText) = myFixture.configureByText("probe.lua", text)

    private fun nameRefs(): List<LuaNameRef> =
        ReadAction.compute<List<LuaNameRef>, RuntimeException> {
            PsiTreeUtil.findChildrenOfType(myFixture.file, LuaNameRef::class.java).toList()
        }

    private fun describe(element: PsiElement?): String =
        if (element == null) {
            "null"
        } else {
            "${element.javaClass.simpleName}@${element.textRange}'${element.text}'"
        }

    private fun offsetOf(
        needle: String,
        occurrence: Int = 1,
    ): Int {
        var index = -1
        repeat(occurrence) { index = fixtureText.indexOf(needle, index + 1) }
        return index
    }

    // ---- premise shared by PsiElement2Declaration / ModCommandExecutorImpl / VcsCodeVisionLanguageContext ----

    fun testProbe01Premise() {
        configure()
        probe("__premise__") {
            ReadAction.run<RuntimeException> {
                for (nameRef in nameRefs()) {
                    val owner = nameRef as? PsiNameIdentifierOwner
                    val identifier = owner?.nameIdentifier
                    emit(
                        "__premise__",
                        "text='${nameRef.text}' isOwner=${owner != null} children=${nameRef.children.size} " +
                            "refRange=${nameRef.textRange} idRange=${identifier?.textRange} " +
                            "rangesEqual=${identifier?.textRange == nameRef.textRange} " +
                            "idIsStrictDescendant=${identifier != null && PsiTreeUtil.isAncestor(nameRef, identifier, true)} " +
                            "isNamed=${nameRef is PsiNamedElement} name=${(nameRef as? PsiNamedElement)?.name}",
                    )
                }
            }
        }
    }

                    emit(

    }

    // ---- SafeDeleteProcessor.isInside ----

    fun testProbe03SafeDelete() {
        probe("SafeDeleteProcessor.isInside") {

                ReadAction.compute<PsiElement?, RuntimeException> {
                    myFixture.file.findElementAt(offsetOf("unused"))

            SafeDeleteHandler.invoke(project, arrayOf(leaf), true)
            emit("SafeDeleteProcessor.isInside", "afterLeafDelete=${myFixture.editor.document.text.replace("\n", "\\n")}")
        }
        probe("SafeDeleteProcessor.isInside") {
            configure()

                ReadAction.compute<PsiElement?, RuntimeException> {
                    myFixture.file.findElementAt(offsetOf("unused"))?.parent
                } ?: return@probe

            SafeDeleteHandler.invoke(project, arrayOf(nameRef), true)
            emit("SafeDeleteProcessor.isInside", "afterNameRefDelete=${myFixture.editor.document.text.replace("\n", "\\n")}")
        }
        probe("SafeDeleteProcessor.isInside") {
            configure()
            val leaf =
                ReadAction.compute<PsiElement?, RuntimeException> {
                    myFixture.file.findElementAt(offsetOf("counter"))
                } ?: return@probe
            SafeDeleteHandler.invoke(project, arrayOf(leaf), true)

        }
    }

    // ---- PsiElement2Declaration.getIdentifyingElement ----

    fun testProbe04PsiElement2Declaration() {
        configure()
        probe("PsiElement2Declaration.getIdentifyingElement") {
            ReadAction.run<RuntimeException> {
                for (label in listOf("counter", "alpha", "helper", "unused")) {
                    val offset = offsetOf(label)
                    val symbols = targetSymbols(myFixture.file, offset)
                    val rendered =
                        symbols.joinToString(";") {
                            val psi = PsiSymbolService.getInstance().extractElementFromSymbol(it)
                            "${it.javaClass.simpleName}->${describe(psi)}"
                        }
                    emit("PsiElement2Declaration.getIdentifyingElement", "at='$label'@$offset count=${symbols.size} [$rendered]")
                }
            }
        }
    }

    // ---- NamedElementDuplicateHandler ----

    fun testProbe05NamedElementDuplicateHandler() {
        // The handler only runs its PsiNameIdentifierOwner branch with NO selection, over the
        // caret's whole LINE (NamedElementDuplicateHandler:44-56), so a sub-name selection can
        // never reach it. These three carets are the lines whose first non-blank element has a
        // LuaNameRef ancestor contained in the line.
        for (line in listOf("counter = helper(1, 2)", "print(counter)", "local counter = 0")) {
            probe("NamedElementDuplicateHandler") {
                configure()
                val lineStart = fixtureText.indexOf(line)
                myFixture.editor.caretModel.moveToOffset(lineStart + line.length)
                myFixture.performEditorAction(IdeActions.ACTION_EDITOR_DUPLICATE)
                emit(
                    "NamedElementDuplicateHandler",
                    "line='$line' caretAfter=${myFixture.editor.caretModel.offset} " +
                        "doc=${myFixture.editor.document.text.replace("\n", "\\n")}",
                )
            }
        }
    }

    // ---- NonAsciiCharactersInspection ----

    fun testProbe07NonAscii() {
        probe("NonAsciiCharactersInspection") {
            configure("local café = 1\nlocal naïve = café\n")
            ReadAction.run<RuntimeException> {
                emit("NonAsciiCharactersInspection", "nameRefCount=${nameRefs().size}")
                for (nameRef in nameRefs()) {
                    emit(
                        "NonAsciiCharactersInspection",
                        "nameRef='${nameRef.text}' allAscii=${nameRef.text.all { it.code < 128 }} id=${describe((nameRef as? PsiNameIdentifierOwner)?.nameIdentifier)}",
                    )
                }
                var leaf = PsiTreeUtil.getDeepestFirst(myFixture.file)
                val types = StringBuilder()
                while (true) {
                    types.append(leaf.node.elementType).append("['").append(leaf.text.replace("\n", "\\n")).append("'] ")
                    leaf = PsiTreeUtil.nextLeaf(leaf) ?: break
                }
                emit("NonAsciiCharactersInspection", "leaves=$types")
            }
        }
    }

    // ---- RelatedItemLineMarkerProvider (row PROMOTED: Lunar registers LuaOverrideLineMarkerProvider) ----

    fun testProbe09RelatedItemGoto() {
        configure(
            "---@class Base\nlocal Base = {}\nfunction Base:run() end\n" +
                "---@class Impl : Base\nlocal Impl = {}\nfunction Impl:run() end\n",
        )
        probe("RelatedItemLineMarkerProvider") {
            ReadAction.run<RuntimeException> {
                val adapter = RelatedItemLineMarkerGotoAdapter()
                val offset = myFixture.file.text.lastIndexOf("run")
                val leaf = myFixture.file.findElementAt(offset)
                val leafItems = if (leaf == null) emptyList() else adapter.getItems(leaf)
                emit("RelatedItemLineMarkerProvider", "fromLeaf n=${leafItems.size} ${leafItems.joinToString(";") { describe(it.element) }}")
                val parent = leaf?.parent
                val parentItems = if (parent == null) emptyList() else adapter.getItems(parent)
                emit("RelatedItemLineMarkerProvider", "fromNameRef n=${parentItems.size} ${parentItems.joinToString(";") { describe(it.element) }}")
                myFixture.editor.caretModel.moveToOffset(offset)
                val editorItems = GotoRelatedSymbolAction.getItems(myFixture.file, myFixture.editor)
                emit(
                    "RelatedItemLineMarkerProvider",
                    "viaEditorAction n=${editorItems.size} ${editorItems.joinToString(";") { describe(it.element) }}",
                )
            }
        }
    }

    // ---- Spellchecker family (NOT in design §4; found by re-enumerating the compiled platform) ----

    fun testProbe10Spellchecker() {
        configure()
        probe("SpellcheckingStrategy.getTokenizer") {
            ReadAction.run<RuntimeException> {
                for (nameRef in nameRefs()) {
                    val strategy = SpellcheckingStrategy.getSpellcheckingStrategy(nameRef)
                    emit(
                        "SpellcheckingStrategy.getTokenizer",
                        "on='${nameRef.text}' strategy=${strategy?.javaClass?.simpleName} tokenizer=${strategy?.getTokenizer(nameRef)?.javaClass?.simpleName}",
                    )
                }
            }
        }
        probe("RenameTo") {
            val nameRef =
                ReadAction.compute<LuaNameRef?, RuntimeException> {
                    nameRefs().firstOrNull { it.text == "counter" }
                } ?: return@probe
            ReadAction.run<RuntimeException> {
                val named = PsiTreeUtil.getNonStrictParentOfType(nameRef, PsiNamedElement::class.java)
                emit("RenameTo", "namedParent=${describe(named)} name=${named?.name}")
            }
            val pointer = SmartPointerManager.getInstance(project).createSmartPsiElementPointer(nameRef, myFixture.file)
            emit("RenameTo", "smartPointerElement=${describe(pointer.element)}")
            val fix = RenameTo("conter", TextRange(0, 6), nameRef, SpellcheckerRateTracker(nameRef))
            val available = fix.isAvailable(project, myFixture.editor, myFixture.file)
            emit("RenameTo", "onNameRef isAvailable=$available name='${if (available) fix.name else "n/a"}'")
            val inRead =
                ReadAction.compute<Boolean, RuntimeException> {
                    RenameTo("conter", TextRange(0, 6), nameRef, SpellcheckerRateTracker(nameRef))
                        .isAvailable(project, myFixture.editor, myFixture.file)
                }
            emit("RenameTo", "onNameRefInsideReadAction isAvailable=$inRead")
            emit("RenameTo", "nameRelativeRange=${nameRelativeRangeOf(fix, nameRef)}")
        }
        probe("RenameTo") {
            val leaf =
                ReadAction.compute<PsiElement?, RuntimeException> {
                    myFixture.file.findElementAt(offsetOf("counter"))
                } ?: return@probe
            val fix = RenameTo("conter", TextRange(0, 6), leaf, SpellcheckerRateTracker(leaf))
            val available = fix.isAvailable(project, myFixture.editor, myFixture.file)
            emit("RenameTo", "onLeaf isAvailable=$available name='${if (available) fix.name else "n/a"}'")
            val named = ReadAction.compute<PsiNamedElement?, RuntimeException> { leaf.parent as? PsiNamedElement }
            emit("RenameTo", "onLeaf nameRelativeRange=${if (named == null) "no-named-parent" else nameRelativeRangeOf(fix, named)}")
        }
    }

        }

    /**
     * Reads the private `getNameRelativeRange` — the frame carrying the shipped build's
     * `PsiNameIdentifierOwner` branch (verified in bytecode at offset 16-20). Reflection is used
     * because the frame is private and reachable only via `isAvailable`, which the shipped build
     * additionally gates on a non-empty suggestion list that the offline test dictionary never fills.
     */
    private fun nameRelativeRangeOf(
        fix: RenameTo,
        named: PsiNamedElement,
    ): String =
        try {
            val method = RenameTo::class.java.getDeclaredMethod("getNameRelativeRange", PsiNamedElement::class.java)
            method.isAccessible = true
            ReadAction.compute<String, RuntimeException> { "${method.invoke(fix, named)}" }
        } catch (failure: ReflectiveOperationException) {
            "REFLECTION-FAILED=${failure.javaClass.simpleName}: ${failure.message}"
        }
}
