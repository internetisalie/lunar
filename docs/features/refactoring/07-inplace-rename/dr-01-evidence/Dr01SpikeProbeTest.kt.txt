class Dr01SpikeProbeTest : BasePlatformTestCase() {
    @Test
    fun testProbeAnameIdentifier() {
        myFixture.configureByText("dr01a.lua", DECL_FIXTURE)
        runReadAction {
            val supplied = PsiElementRenameHandler.getElement(contextOf())
            emit("a", "supplied.javaClass", supplied?.javaClass?.name)
            emit("a", "supplied is PsiNameIdentifierOwner", (supplied is PsiNameIdentifierOwner).toString())
            val owner = supplied as? PsiNameIdentifierOwner
            val identifier = owner?.nameIdentifier
            emit("a", "nameIdentifier.javaClass", identifier?.javaClass?.name)
            emit("a", "nameIdentifier.elementType", identifier?.node?.elementType?.toString())
            emit("a", "nameIdentifier.textRange", identifier?.textRange?.toString())
            emit("a", "nameIdentifier.text", identifier?.text)
            emit("a", "leafAtCaret", describeLeafAtCaret())
            emit("a", "nameIdentifier === leafAtCaret", (identifier === myFixture.file.findElementAt(myFixture.caretOffset)).toString())
        }
    }

    @Test
    fun testProbeBgates() {
        CARET_SHAPES.forEach { shape ->
            myFixture.configureByText("dr01b.lua", shape.second)
            val context = contextOf()
            runReadAction {
                val supplied = PsiElementRenameHandler.getElement(context)
                emit(shape.first, "supplied.javaClass", supplied?.javaClass?.name)
                emit(shape.first, "supplied.elementType", supplied?.node?.elementType?.toString())
                emit(shape.first, "supplied.textRange", supplied?.textRange?.toString())
                emit(shape.first, "supplied.text", supplied?.text)
                emit(shape.first, "supplied is PsiNameIdentifierOwner", (supplied is PsiNameIdentifierOwner).toString())
                emit(shape.first, "supplied is LuaNameRef", (supplied is LuaNameRef).toString())
                emit(shape.first, "leafAtCaret", describeLeafAtCaret())
            }
            emit(shape.first, "MemberInplaceRenameHandler.isAvailableOnDataContext", guarded { MemberInplaceRenameHandler().isAvailableOnDataContext(context).toString() })
            emit(shape.first, "LuaInplaceRenameHandler.isAvailableOnDataContext", guarded { LuaInplaceRenameHandler().isAvailableOnDataContext(context).toString() })
            emit(shape.first, "RenameHandlerRegistry.getRenameHandlers", guarded { handlersOf(context) })
            emit(shape.first, "EP handlers whose isRenaming is true", guarded { renamingExtensionsOf(context) })
            emit(shape.first, "RenameHandler.EP_NAME size", RenameHandler.EP_NAME.extensionList.size.toString())
        }
    }

    @Test
    fun testProbeB2usageCaretCommits() {
        myFixture.configureByText("dr01b2.lua", "local counter = 0\nprint(coun<caret>ter)\ncounter = counter + 1\n")
        val started = renameInPlaceViaHandler(LuaInplaceRenameHandler(), "total")
        emit("b2", "renameInPlaceViaHandler returned", started.toString())
        emit("b2", "document", myFixture.editor.document.text.replace("\n", "\\n"))
    }

    @Test
    fun testProbeCandD() {
        myFixture.configureByText("dr01c.lua", DECL_FIXTURE)
        val disposable = Disposer.newDisposable()
        try {
            TemplateManagerImpl.setTemplateTesting(disposable)
            val nameRef = runReadAction { PsiElementRenameHandler.getElement(contextOf()) }
            emit("c", "elementToRename.javaClass", nameRef?.javaClass?.name)
            val leaf = runReadAction { myFixture.file.findElementAt(myFixture.caretOffset) }
            val named = nameRef as? PsiNamedElement
            if (named == null) {
                emit("c", "performInplaceRename", "NOT-RUN: supplied element is not a PsiNamedElement")
                return
            }
            val renamer = MemberInplaceRenamer(named, leaf, myFixture.editor)
            val returned = guarded { renamer.performInplaceRename(null).toString() }
            emit("c", "performInplaceRename(null)", returned)
            val state = TemplateManagerImpl.getTemplateState(myFixture.editor)
            emit("d", "templateState", if (state == null) "null" else state.javaClass.name)
            emit("d", "currentVariableRange", state?.currentVariableRange?.toString())
            recordSegments("d", state)
            state?.gotoEnd(true)
        } finally {
            Disposer.dispose(disposable)
        }
    }

    fun testProbeEdocumentAfterTryInlineRename() {

            CodeInsightTestUtil.tryInlineRename(
                MemberInplaceRenameHandler(),
                "total",

                requireNotNull(leaf) { "no leaf at caret" },

        emit("e", "tryInlineRename returned", started.toString())

    }

    @Test
    fun testProbeFparameterCaret() {
        myFixture.configureByText("dr01f.lua", "---@param a number\nlocal function f(<caret>a) return a end\n")
        val disposable = Disposer.newDisposable()
        try {
            TemplateManagerImpl.setTemplateTesting(disposable)
            val context = contextOf()
            LuaInplaceRenameHandler().invoke(project, myFixture.editor, myFixture.file, context)
            val state = TemplateManagerImpl.getTemplateState(myFixture.editor)
            emit("f", "templateState", if (state == null) "null" else state.javaClass.name)
            emit("f", "currentVariableRange", state?.currentVariableRange?.toString())
            recordSegments("f", state)
            if (state != null) {
                val range = requireNotNull(state.currentVariableRange)
                WriteCommandAction.writeCommandAction(project).run<RuntimeException> {
                    myFixture.editor.document.replaceString(range.startOffset, range.endOffset, "count")
                }
                requireNotNull(TemplateManagerImpl.getTemplateState(myFixture.editor)).gotoEnd(false)
                NonBlockingReadActionImpl.waitForAsyncTaskCompletion()
            }
            emit("f", "document", myFixture.editor.document.text.replace("\n", "\\n"))
        } finally {
            Disposer.dispose(disposable)
        }
    }

    @Test
    fun testProbeGsubstitutionCallback() {
        myFixture.configureByText("dr01g.lua", DECL_FIXTURE)
        val disposable = Disposer.newDisposable()
        try {
            TemplateManagerImpl.setTemplateTesting(disposable)
            val context = contextOf()
            val nameRef = runReadAction { PsiElementRenameHandler.getElement(context) }
            emit("g", "elementToRename.javaClass", nameRef?.javaClass?.name)
            recordDirectSubstitution(nameRef)
            if (nameRef != null) {
                MemberInplaceRenameHandler().doRename(nameRef, myFixture.editor, context)
            }
            val state = TemplateManagerImpl.getTemplateState(myFixture.editor)
            emit("g", "doRename -> templateState", if (state == null) "null" else state.javaClass.name)
            emit("g", "doRename -> currentVariableRange", state?.currentVariableRange?.toString())
            state?.gotoEnd(true)
        } finally {
            Disposer.dispose(disposable)
        }
    }

    private fun recordDirectSubstitution(elementToRename: PsiElement?) {
        val target = elementToRename ?: return
        val processor = runReadAction { RenamePsiElementProcessor.forElement(target) }
        emit("g", "RenamePsiElementProcessor.forElement", processor.javaClass.name)
        emit("g", "isInplaceRenameSupported", processor.isInplaceRenameSupported.toString())
        emit("g", "substituteElementToRename(2-arg)", guarded { runReadAction { processor.substituteElementToRename(target, myFixture.editor)?.let { "${it.javaClass.simpleName} ${it.node?.elementType} ${it.textRange}" } ?: "null" } })
        val fired = StringBuilder("NEVER-FIRED")
        val outcome =
            guarded {
                processor.substituteElementToRename(
                    target,
                    myFixture.editor,
                    object : Pass<PsiElement>() {
                        override fun pass(passed: PsiElement) {
                            fired.setLength(0)
                            fired.append(runReadAction { "FIRED ${passed.javaClass.simpleName} ${passed.node?.elementType} ${passed.textRange}" })
                        }
                    },
                )
                "returned-normally"
            }
        emit("g", "substituteElementToRename(3-arg) call", outcome)
        emit("g", "substitution callback", fired.toString())
    }

    private fun recordSegments(
        label: String,
        state: TemplateState?,
    ) {
        val live = state ?: return
        val template = live.template ?: return
        (0 until template.variableCount).forEach { index ->
            val name = template.getVariableNameAt(index)
            emit(label, "variable[$index]", "$name range=${live.getVariableRange(name)}")
        }
    }

    private fun renameInPlaceViaHandler(
        handler: RenameHandler,
        newName: String,
    ): Boolean {
        val disposable = Disposer.newDisposable()
        try {
            TemplateManagerImpl.setTemplateTesting(disposable)
            val context = DataManager.getInstance().getDataContext(myFixture.editor.contentComponent)
            handler.invoke(project, myFixture.editor, myFixture.file, context)
            val started = TemplateManagerImpl.getTemplateState(myFixture.editor) ?: return false
            val range = requireNotNull(started.currentVariableRange) { "template started with no current variable" }
            WriteCommandAction.writeCommandAction(project).run<RuntimeException> {
                myFixture.editor.document.replaceString(range.startOffset, range.endOffset, newName)
            }
            requireNotNull(TemplateManagerImpl.getTemplateState(myFixture.editor)).gotoEnd(false)
            NonBlockingReadActionImpl.waitForAsyncTaskCompletion()
        } finally {
            Disposer.dispose(disposable)
        }
        return true
    }

    private fun contextOf(): DataContext = DataManager.getInstance().getDataContext(myFixture.editor.contentComponent)

    private fun handlersOf(context: DataContext): String =
        RenameHandlerRegistry
            .getInstance()
            .getRenameHandlers(context)
            .joinToString(", ") { handler -> handler.javaClass.name }
            .ifEmpty { "EMPTY-LIST" }

    private fun renamingExtensionsOf(context: DataContext): String =
        RenameHandler.EP_NAME.extensionList
            .filter { handler -> handler.isRenaming(context) }
            .joinToString(", ") { handler -> handler.javaClass.name }
            .ifEmpty { "EMPTY-LIST" }

    private fun describeLeafAtCaret(): String {
        val leaf = myFixture.file.findElementAt(myFixture.caretOffset) ?: return "null"
        return "${leaf.javaClass.simpleName}(${leaf.node?.elementType}) '${leaf.text}' ${leaf.textRange}"
    }

    private fun guarded(probe: () -> String): String =
        runCatching(probe).getOrElse { failure -> "THREW ${failure.javaClass.name}: ${failure.message}" }

    private fun emit(
        label: String,
        field: String,
        value: String?,
    ) {
        println("DR01|$label|$field|${value ?: "null"}")
    }

    private companion object {
        const val DECL_FIXTURE = "local coun<caret>ter = 0\nprint(counter)\ncounter = counter + 1\n"

        val CARET_SHAPES =
            listOf(
                "b1-local-decl" to DECL_FIXTURE,
                "b2-usage" to "local counter = 0\nprint(coun<caret>ter)\ncounter = counter + 1\n",
                "b3-parameter-decl" to "local function f(<caret>a) return a end\n",
                "b4-generic-for-var" to "for k<caret>, v in pairs(t) do\n  print(k, v)\nend\n",
                "b5-local-function-name" to "local function hel<caret>per() return 1 end\nprint(helper())\n",
                "b6-local-shadowing-local" to "local config = 1\nlocal con<caret>fig = 2\nprint(config)\n",
            )
    }
}
