package net.internetisalie.lunar.refactoring.rename

import com.intellij.codeInsight.template.impl.TemplateManagerImpl
import com.intellij.ide.DataManager
import com.intellij.openapi.application.impl.NonBlockingReadActionImpl
import com.intellij.openapi.application.runReadAction
import com.intellij.openapi.command.WriteCommandAction
import com.intellij.openapi.util.Disposer
import com.intellij.psi.PsiDocumentManager
import com.intellij.refactoring.listeners.RefactoringEventData
import com.intellij.refactoring.listeners.RefactoringEventListener
import com.intellij.refactoring.rename.PsiElementRenameHandler
import com.intellij.refactoring.rename.inplace.InplaceRefactoring
import com.intellij.testFramework.PlatformTestUtil
import com.intellij.testFramework.fixtures.BasePlatformTestCase
import org.junit.Test
import org.junit.runner.RunWith
import org.junit.runners.JUnit4

@RunWith(JUnit4::class)
class Dr01SpikeProbe2Test : BasePlatformTestCase() {
    @Test
    fun testParameterCaretWithFullDrain() {
        myFixture.configureByText("dr01f2.lua", PARAM_FIXTURE)
        driveAndRecord("f2", "count")
    }

    @Test
    fun testLocalDeclarationCaretRefactoringEvents() {
        myFixture.configureByText("dr01f3.lua", "local coun<caret>ter = 0\nprint(counter)\n")
        driveAndRecord("f3", "total")
    }

    @Test
    fun testDialogRouteControlOnSameFixture() {
        myFixture.configureByText("dr01f4.lua", PARAM_FIXTURE)
        val events = subscribeRefactoringEvents("f4")
        myFixture.renameElementAtCaret("count")
        drain()
        emit("f4", "document", myFixture.editor.document.text.replace("\n", "\\n"))
        emit("f4", "refactoring events", events.joinToString(" ; ").ifEmpty { "NONE" })
    }

    @Test
    fun testGenericForCaretInsideIdentifier() {
        listOf(
            "b4b-generic-for-first-var-inside" to "for ke<caret>y, value in pairs(t) do\n  print(key, value)\nend\n",
            "b4c-generic-for-second-var-inside" to "for key, val<caret>ue in pairs(t) do\n  print(key, value)\nend\n",
        ).forEach { shape ->
            myFixture.configureByText("dr01b4.lua", shape.second)
            val context = DataManager.getInstance().getDataContext(myFixture.editor.contentComponent)
            runReadAction {
                val supplied = PsiElementRenameHandler.getElement(context)
                emit(shape.first, "supplied.javaClass", supplied?.javaClass?.name)
                emit(shape.first, "supplied.elementType", supplied?.node?.elementType?.toString())
                emit(shape.first, "supplied.textRange", supplied?.textRange?.toString())
            }
            emit(shape.first, "MemberInplaceRenameHandler.isAvailableOnDataContext", memberGate(context))
            emit(shape.first, "LuaInplaceRenameHandler.isAvailableOnDataContext", LuaInplaceRenameHandler().isAvailableOnDataContext(context).toString())
        }
    }

    private fun memberGate(context: com.intellij.openapi.actionSystem.DataContext): String =
        com.intellij.refactoring.rename.inplace
            .MemberInplaceRenameHandler()
            .isAvailableOnDataContext(context)
            .toString()

    private fun driveAndRecord(
        label: String,
        newName: String,
    ) {
        val events = subscribeRefactoringEvents(label)
        val disposable = Disposer.newDisposable()
        try {
            TemplateManagerImpl.setTemplateTesting(disposable)
            val context = DataManager.getInstance().getDataContext(myFixture.editor.contentComponent)
            LuaInplaceRenameHandler().invoke(project, myFixture.editor, myFixture.file, context)
            val state = TemplateManagerImpl.getTemplateState(myFixture.editor)
            emit(label, "templateState", if (state == null) "null" else "present")
            val range = state?.currentVariableRange
            emit(label, "currentVariableRange", range?.toString())
            if (range == null) return
            WriteCommandAction.writeCommandAction(project).run<RuntimeException> {
                myFixture.editor.document.replaceString(range.startOffset, range.endOffset, newName)
            }
            emit(label, "document before gotoEnd", myFixture.editor.document.text.replace("\n", "\\n"))
            requireNotNull(TemplateManagerImpl.getTemplateState(myFixture.editor)).gotoEnd(false)
            emit(label, "document right after gotoEnd", myFixture.editor.document.text.replace("\n", "\\n"))
            drain()
            emit(label, "document after full drain", myFixture.editor.document.text.replace("\n", "\\n"))
            emit(label, "INPLACE_RENAMER user data", myFixture.editor.getUserData(InplaceRefactoring.INPLACE_RENAMER)?.javaClass?.name)
        } finally {
            Disposer.dispose(disposable)
        }
        emit(label, "refactoring events", events.joinToString(" ; ").ifEmpty { "NONE" })
        emit(label, "document after dispose", myFixture.editor.document.text.replace("\n", "\\n"))
    }

    private fun drain() {
        repeat(5) {
            NonBlockingReadActionImpl.waitForAsyncTaskCompletion()
            PlatformTestUtil.dispatchAllInvocationEventsInIdeEventQueue()
            PsiDocumentManager.getInstance(project).commitAllDocuments()
        }
    }

    private fun subscribeRefactoringEvents(label: String): MutableList<String> {
        val seen = mutableListOf<String>()
        project.messageBus
            .connect(testRootDisposable)
            .subscribe(
                RefactoringEventListener.REFACTORING_EVENT_TOPIC,
                object : RefactoringEventListener {
                    override fun refactoringStarted(
                        refactoringId: String,
                        beforeData: RefactoringEventData?,
                    ) {
                        seen.add("started:$refactoringId")
                    }

                    override fun refactoringDone(
                        refactoringId: String,
                        afterData: RefactoringEventData?,
                    ) {
                        seen.add("done:$refactoringId")
                    }
                },
            )
        emit(label, "listener subscribed", "true")
        return seen
    }

    private fun emit(
        label: String,
        field: String,
        value: String?,
    ) {
        println("DR01|$label|$field|${value ?: "null"}")
    }

    private companion object {
        const val PARAM_FIXTURE = "---@param a number\nlocal function f(<caret>a) return a end\n"
    }
}
